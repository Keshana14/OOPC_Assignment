class Job
! Represents a repair record. !

uses 
    Vehicle with 1 label repairs  ! Vehicle being repaired. !;
    Mechanic with 1 label assigned ! Assigned mechanic. !;
end

services
        calculateCost  ! Calculate total repair costs. !;
        calculateTime  ! Calculate completion time. !;
        generateRecord ! Create job record. !;
        status  ! Return job status  !;
end

uses
    Vehicle with 0 to * label contains ! Waiting vehicles. !;
end

        addVehicle ! Add vehicle to waiting list. !;
        removeVehicle ! Remove vehicle from the waiting list. !;
        nextVehicle  ! Get the next waiting vehicle. !;

end
